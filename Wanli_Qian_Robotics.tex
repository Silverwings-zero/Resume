\section*{Research Experience}
\role{USC HaRVI Lab | Ph.D. Researcher}{2024--Present}
\begin{itemize}
\item Build bilateral teleoperation systems with tactile feedback for remote surface perception, manipulation, and data collection.
\item Study active haptic perception through exploratory policies that adapt robot actions to reduce uncertainty about object and surface properties.
\item Develop language-conditioned generative models for tactile and vibration signals using force, acceleration, and vision-based tactile data.
\item Curate multimodal haptic datasets spanning 200+ materials, combining tapping, sliding, audio, and affective descriptors.
\item Integrate VR/AR and language-based interaction into haptic and texture authoring systems.
\end{itemize}
\vspace{3pt}
\role{University of Chicago AxLab | Predoctoral Researcher}{2022--2024}
\begin{itemize}
\item Led the development of SHAPE-IT into an interactive system for conversational 3D shape generation and real-time reconfiguration of physical shape displays.
\item Developed adaptive LLM-based agents that modify shape-display control scripts in response to ongoing interaction.
\end{itemize}
\vspace{3pt}
\role{Georgia Tech Borg Lab | Lab Assistant}{2021--2022}
